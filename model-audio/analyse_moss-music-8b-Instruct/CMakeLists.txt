cmake_minimum_required(VERSION 3.21)
project(moss-music-cli LANGUAGES C CXX)

# 直接编译固定上游的原版 CLI；不复制第三方源码，也不引入其完整音乐生成/VST 工程。
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MOSS_MUSIC_SOURCE_DIR "" CACHE PATH "HOT-Step-CPP 源码目录")
set(ENGINE_DIR "${MOSS_MUSIC_SOURCE_DIR}/engine")
if(NOT EXISTS "${ENGINE_DIR}/ggml/include/ggml.h")
    message(FATAL_ERROR "请先准备 HOT-Step-CPP 的 engine/ggml 固定子模块")
endif()
add_compile_definitions(GGML_MAX_NAME=128)
set(GGML_CUDA ON CACHE BOOL "使用 CUDA 后端")
set(GGML_BACKEND_DL OFF CACHE BOOL "显式链接后端")
set(GGML_BUILD_TESTS OFF CACHE BOOL "不编译 GGML 测试")
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "不编译 GGML 示例")
add_subdirectory("${ENGINE_DIR}/ggml" ggml)
find_package(Threads REQUIRED)
add_library(moss-json STATIC "${ENGINE_DIR}/vendor/yyjson/yyjson.c")
target_include_directories(moss-json PUBLIC "${ENGINE_DIR}/vendor/yyjson")
add_executable(ace-caption "${ENGINE_DIR}/tools/ace-caption.cpp")
target_include_directories(ace-caption PRIVATE "${ENGINE_DIR}" "${ENGINE_DIR}/src")
target_link_libraries(ace-caption PRIVATE ggml moss-json Threads::Threads ${CMAKE_DL_LIBS})
foreach(backend cpu cuda)
    if(TARGET ggml-${backend})
        target_link_libraries(ace-caption PRIVATE ggml-${backend})
    endif()
endforeach()
